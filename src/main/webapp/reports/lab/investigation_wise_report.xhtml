function hidePaginatorBeforePrint() { let paginators = document.querySelectorAll('.ui-paginator'); paginators.forEach(el => el.style.display = 'none'); } function restorePaginatorAfterPrint() { let paginators = document.querySelectorAll('.ui-paginator'); paginators.forEach(el => el.style.display = ''); } window.addEventListener('afterprint', restorePaginatorAfterPrint);
