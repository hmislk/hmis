function captureChartImages() { try { var barWidget = PF('surgeryCountBarWV'); var barInput = document.querySelector('input[id$=":barChartImageData"]'); if (barWidget && barWidget.chart && barInput) { barInput.value = barWidget.chart.toBase64Image('image/png', 1); } else if (!barInput) { console.warn('barChartImageData hidden field not found in DOM'); } } catch (e) { console.warn('Bar chart capture failed', e); } try { var lineWidget = PF('surgeryCountLineWV'); var lineInput = document.querySelector('input[id$=":lineChartImageData"]'); if (lineWidget && lineWidget.chart && lineInput) { lineInput.value = lineWidget.chart.toBase64Image('image/png', 1); } else if (!lineInput) { console.warn('lineChartImageData hidden field not found in DOM'); } } catch (e) { console.warn('Line chart capture failed', e); } }
