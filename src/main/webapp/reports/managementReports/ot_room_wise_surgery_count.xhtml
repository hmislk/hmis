function captureChartsForDownload() { var pairs = [ { wv: 'expOtRoomBarWV', id: 'otRoomForm:otRoomBarChartImg' }, { wv: 'expOtRoomLineWV', id: 'otRoomForm:otRoomLineChartImg' } ]; pairs.forEach(function(pair) { try { var widget = PF(pair.wv); if (widget && widget.chart) { var input = document.getElementById(pair.id); if (input) { input.value = widget.chart.toBase64Image('image/png', 1); } } } catch(e) { console.warn('Chart export capture failed: ' + pair.wv, e); } }); }
