function captureChartsForDownload() { var pairs = [ { wv: 'expOtRoomBarWV', id: 'otRoomForm:otRoomBarChartImg' }, { wv: 'expOtRoomLineWV', id: 'otRoomForm:otRoomLineChartImg' } ]; pairs.forEach(function(pair) { try { var widget = PF(pair.wv); if (widget && widget.chart) { var input = document.getElementById(pair.id); if (input) { input.value = widget.chart.toBase64Image('image/png', 1); } } } catch(e) { console.warn('Chart export capture failed: ' + pair.wv, e); } }); } // p:printer clones the printArea DOM into a separate print window, but a // <canvas> clone carries no pixels - only the <img> below actually // prints. Swap the on-screen chart for a captured PNG right before printing, // then swap back after the print dialog has had time to read the DOM. function captureChartsForPrint() { var pairs = [ { wv: 'otRoomBarWV', imgId: 'otRoomBarChartPrintImg', wrapId: 'otRoomBarChartCanvasWrap' }, { wv: 'otRoomLineWV', imgId: 'otRoomLineChartPrintImg', wrapId: 'otRoomLineChartCanvasWrap' } ]; pairs.forEach(function(pair) { try { var widget = PF(pair.wv); var img = document.getElementById(pair.imgId); var wrap = document.getElementById(pair.wrapId); if (widget && widget.chart && img && wrap) { img.src = widget.chart.toBase64Image('image/png', 1); img.style.display = 'block'; wrap.style.display = 'none'; setTimeout(function() { img.style.display = 'none'; wrap.style.display = ''; }, 2000); } } catch(e) { console.warn('Chart print capture failed: ' + pair.wv, e); } }); }
