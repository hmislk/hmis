function convertRouteAnalysisChartToImage(widgetVarName) { return new Promise(function (resolve) { var chartWidget = PF(widgetVarName); if (!chartWidget || !chartWidget.cfg || !chartWidget.cfg.chart) { resolve(); return; } var chart = chartWidget.cfg.chart; var canvas = chart.canvas; if (!canvas || canvas.tagName !== 'CANVAS') { resolve(); return; } var image = document.createElement('img'); image.className = canvas.className; image.style.cssText = canvas.style.cssText; image.style.width = '100%'; image.style.height = canvas.style.height || '70mm'; image.onload = function () { if (canvas.parentNode) { canvas.parentNode.replaceChild(image, canvas); } resolve(); }; image.onerror = function () { resolve(); }; image.src = chart.toBase64Image(); }); } function printRouteAnalysisPreview() { Promise.all([ convertRouteAnalysisChartToImage('printCountChartWidget'), convertRouteAnalysisChartToImage('printServiceChartWidget') ]).then(function () { setTimeout(function () { window.print(); }, 300); }); }
